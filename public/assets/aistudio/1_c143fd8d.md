**<span dir="rtl">اصول سه‌گانه</span> Truth Maintenance
<span dir="rtl">شرط‌مند و دو-زمانه</span>**

<span dir="rtl">موتور</span> Delta-Merge <span dir="rtl">برای افزودن
دانش جدید بدون تخریب دانش قبلی و بدون خلق تعارض‌های کاذب، باید بر سه
الگوی پژوهشی هم‌راستا استوار شود. نخست، الگوی **گراف دو-زمانه
(**</span>**Bi-Temporal<span dir="rtl">)</span>** <span dir="rtl">در
معماری</span> Graphiti/Zep <span dir="rtl">است که زمان اعتبار واقعی یک
گزاره در دنیای مهندسی (</span>valid_from <span dir="rtl">و</span>
valid_to<span dir="rtl">، مثلاً بازه تولید خودرو یا نسخه نرم‌افزار</span>
ECU<span dir="rtl">) را از زمان یادگیری و ثبت در سامانه
(</span>ingested_at <span dir="rtl">و</span>
invalidated_at<span dir="rtl">) تفکیک می‌کند. هنگامی که گزاره‌ای منسوخ یا
اصلاح می‌شود، رکورد قبلی حذف نمی‌شود بلکه با ثبت</span> invalidated_at
<span dir="rtl">و ارجاع به رکورد جایگزین
(</span>superseded_by<span dir="rtl">) از چرخه دانش فعال خارج می‌شود تا
تاریخچه حسابرسی و قابلیت بازگشت به نسخه پیشین حفظ گردد. برخلاف</span>
Graphiti <span dir="rtl">که باطل‌سازی یال‌های متناقض را به</span> LLM
<span dir="rtl">می‌سپارد، در یک پلتفرم حساس مهندسی خودرو باید باطل‌سازی و
جایگزینی از مسیر **قواعد قطعی
(**</span>**Rule-Based<span dir="rtl">)</span>** <span dir="rtl">و
سلسله‌مراتب اعتبار شواهد عبور کند تا توهم مدل زبانی باعث حذف ناخواسته
داده‌های کارخانه‌ای نشود.</span>

<span dir="rtl">دوم، الگوی **جداسازی هسته نتیجه‌گیری از شواهد
(**</span>**KB vs. Evidence<span dir="rtl">)</span>**
<span dir="rtl">برگرفته از معماری</span> Knosis <span dir="rtl">است. در
این الگو، جدول گزاره‌های کتاب مرجع فقط نتیجه‌گیری ساختاریافته نهایی
(پارامتر، مقدار، واحد و شرط‌ها) را نگه می‌دارد و تمام منشأهای استخراج،
لاگ‌های داینو، فایل‌های</span> A2L/Damos <span dir="rtl">و گزارش‌های گارد
لایه ۱ در جدول جداگانه شواهد (</span>evidence_records<span dir="rtl">)
ذخیره می‌شوند. با این جداسازی، یک گزاره معتبر می‌تواند هم‌زمان به یک
فایل</span> A2L <span dir="rtl">رسمی و چند لاگ میدانی هم‌راستا متصل شود و
ضریب اطمینان آن بدون تکثیر سطرهای متضاد در کتاب افزایش یابد.</span>

<span dir="rtl">سوم، الگوی **آشتی محافظه‌کارانه قیدهای شرطی
(**</span>**Conservative Qualifier
Reconciliation<span dir="rtl">)</span>** <span dir="rtl">است که بر مبنای

فیزیکی از تلورانس مجاز خارج شوند، تعارض واقعی رخ می‌دهد.</span>

export type EvidenceTier =

\| 'TIER_1_OEM_A2L_DAMOS'

\| 'TIER_2_LAB_DYNO_LOG'

\| 'TIER_3_VERIFIED_MANUAL'

\| 'TIER_4_FIELD_EXPERIENCE';

const TIER_WEIGHT: Record\<EvidenceTier, number\> = {

TIER_1_OEM_A2L_DAMOS: 1.0,

TIER_2_LAB_DYNO_LOG: 0.85,

TIER_3_VERIFIED_MANUAL: 0.7,

TIER_4_FIELD_EXPERIENCE: 0.4,

};

const DISCRIMINATING_KEYS = \[

'brand',

'model',

'ecu_hw',

'ecu_sw',

'fuel_octane_ron',

'fuel_type',

'emission_std',

'hardware_state',

'year_range',

\] as const;

export interface ContextScope {

engine_code: string;

brand?: string;

model?: string;

ecu_hw?: string;

ecu_sw?: string;

fuel_octane_ron?: number;

fuel_type?: 'GASOLINE' \| 'E85' \| 'DIESEL' \| 'CNG' \| 'HYBRID_PHEV';

emission_std?: string;

hardware_state?: 'STOCK' \| 'STAGE_1' \| 'STAGE_2' \| 'MODIFIED';

year_range?: \[number, number\];

}

export interface ClaimCandidate {

claimId?: string;

subjectId: string;

predicate: string;

scope: ContextScope;

valueNumeric?: number;

valueText?: string;

toleranceAbs: number;

tier: EvidenceTier;

}

export type ScopeRelation =

\| 'IDENTICAL'

\| 'SUBSET_SPECIALIZATION'

\| 'DISJOINT_COMPLEMENT';

export function compareContextScopes(a: ContextScope, b: ContextScope):
ScopeRelation {

if (a.engine_code !== b.engine_code) {

return 'DISJOINT_COMPLEMENT';

}

let aMoreSpecific = false;

let bMoreSpecific = false;

for (const key of DISCRIMINATING_KEYS) {

const va = a\[key\];

const vb = b\[key\];

if (va !== undefined && vb !== undefined) {

if (key === 'year_range') {

const \[aMin, aMax\] = va as \[number, number\];

const \[bMin, bMax\] = vb as \[number, number\];

const overlaps = Math.max(aMin, bMin) \<= Math.min(aMax, bMax);

if (!overlaps) return 'DISJOINT_COMPLEMENT';

if (aMin \> bMin \|\| aMax \< bMax) aMoreSpecific = true;

if (bMin \> aMin \|\| bMax \< aMax) bMoreSpecific = true;

} else if (va !== vb) {

return 'DISJOINT_COMPLEMENT';

}

} else if (va !== undefined && vb === undefined) {

aMoreSpecific = true;

} else if (va === undefined && vb !== undefined) {

bMoreSpecific = true;

}

}

if (aMoreSpecific && !bMoreSpecific) return 'SUBSET_SPECIALIZATION';

if (!aMoreSpecific && bMoreSpecific) return 'SUBSET_SPECIALIZATION';

if (aMoreSpecific && bMoreSpecific) return 'DISJOINT_COMPLEMENT';

return 'IDENTICAL';

}

function areClaimValuesCompatible(a: ClaimCandidate, b: ClaimCandidate):
boolean {

if (a.valueNumeric !== undefined && b.valueNumeric !== undefined) {

const tol = Math.max(a.toleranceAbs, b.toleranceAbs);

return Math.abs(a.valueNumeric - b.valueNumeric) \<= tol;

}

if (a.valueText !== undefined && b.valueText !== undefined) {

return a.valueText.trim().toLowerCase() ===
b.valueText.trim().toLowerCase();

}

return false;

}

export type MergeDecision =

\| { action: 'CORROBORATE_EXISTING'; targetClaimId: string }

\| { action: 'INSERT_CONDITIONAL_COMPLEMENT'; reason: string }

\| { action: 'INSERT_SPECIALIZATION_OVERRIDE'; baseClaimId: string }

\| { action: 'KEEP_OEM_FLAG_CONTESTED_NOTE'; activeClaimId: string;
escalateJudge: false }

\| { action: 'OPEN_HARD_CONFLICT_TICKET'; existingClaimId: string;
escalateJudge: true };

export function resolveDeltaMerge(

existing: ClaimCandidate,

incoming: ClaimCandidate

): MergeDecision {

const scopeRel = compareContextScopes(existing.scope, incoming.scope);

if (scopeRel === 'DISJOINT_COMPLEMENT') {

return {

action: 'INSERT_CONDITIONAL_COMPLEMENT',

reason: 'Context qualifiers differ (e.g., fuel octane, ECU SW, or tune
stage); preserve both.',

};

}

if (areClaimValuesCompatible(existing, incoming)) {

return { action: 'CORROBORATE_EXISTING', targetClaimId:
existing.claimId! };

}

if (scopeRel === 'SUBSET_SPECIALIZATION') {

return {

action: 'INSERT_SPECIALIZATION_OVERRIDE',

baseClaimId: existing.claimId!,

};

}

const wExisting = TIER_WEIGHT\[existing.tier\];

const wIncoming = TIER_WEIGHT\[incoming.tier\];

if (wExisting - wIncoming \>= 0.30) {

return {

action: 'KEEP_OEM_FLAG_CONTESTED_NOTE',

activeClaimId: existing.claimId!,

escalateJudge: false,

};

}

return {

action: 'OPEN_HARD_CONFLICT_TICKET',

existingClaimId: existing.claimId!,

escalateJudge: true,

};

}

BEGIN;

SET LOCAL pg_trgm.word_similarity_threshold = 0.35;

SELECT

ce.entity_id,

ce.canonical_code,

ea.raw_alias,

word_similarity(:normalized_query, ea.normalized_key) AS trgm_score

FROM entity_aliases ea

JOIN canonical_entities ce ON ce.entity_id = ea.entity_id

WHERE :normalized_query \<% ea.normalized_key

ORDER BY trgm_score DESC

LIMIT 5;

COMMIT;

const HEX_ADDR_REGEX = /\b0x\[0-9A-Fa-f\]{4,8}\b/gi;

const DYNO_METRIC_REGEX =
/\b(AFR\|Lambda\|Knock\|Retard\|Boost\|BTDC\|ATDC\|CAN\|UDS\|OBD\|mbar\|kPa\|deg\|RPM)\b/gi;

const CONFLICT_HINT_REGEX = /(<span dir="rtl">تناقض\|مغایرت\|مقایسه\|بین
دو منبع\|کدام صحیح\|داینو\|خطای همزمان\|شبکه مالتی</span>\|CAN)/g;

export function scoreAutomotiveComplexity(signals: RoutingSignals) {

const text = signals.rawText ?? '';

const dtcCount = (text.match(DTC_REGEX) ?? \[\]).length;

const hexCount = (text.match(HEX_ADDR_REGEX) ?? \[\]).length;

const dynoMetricCount = (text.match(DYNO_METRIC_REGEX) ?? \[\]).length;

const hasConflictHints = CONFLICT_HINT_REGEX.test(text);

const totalEntities = dtcCount + hexCount + dynoMetricCount;

const tokenBasis = Math.max(0.5, signals.estimatedInputTokens / 1_000);

const entityDensityPer1kTokens = Number((totalEntities /
tokenBasis).toFixed(2));

let score = 0;

score += Math.min(dtcCount \* 0.12, 0.36);

score += Math.min(hexCount \* 0.08, 0.24);

score += Math.min(dynoMetricCount \* 0.06, 0.24);

if (entityDensityPer1kTokens \>= 3.0) score += 0.15;

if (hasConflictHints) score += 0.2;

if (signals.estimatedInputTokens \> 40_000) score += 0.12;

if (signals.hasImagesOrDiagrams) score += 0.1;

if (signals.safetyCritical) score = Math.max(score, 0.85);

return {

complexityScore: Math.min(1, Number(score.toFixed(2))),

entityDensityPer1kTokens,

detectedEntities: { dtcCount, hexCount, dynoMetricCount },

};

}

function isProviderHealthy(

provider: AiProviderId,

healthMap: Record\<AiProviderId, ProviderHealthStatus\>,

nowMs: number

): boolean {

const status = healthMap\[provider\];

if (!status) return true;

if (status.state === 'OPEN') {

return status.cooldownUntilMs !== undefined && nowMs \>=
status.cooldownUntilMs;

}

return status.error429Rate \< 0.25 && status.error5xxRate \< 0.15;

}

export function estimatePlanCostUsd(

entry: ModelCatalogEntry,

signals: RoutingSignals,

useBatchApi: boolean

): number {

let inRate = entry.inputPer1M;

let outRate = entry.outputPer1M;

const cacheRate = entry.cachedInputPer1M ?? inRate;

const effTotalIn = Math.ceil(signals.estimatedInputTokens \*
entry.tokenizerMultiplier);

const effCachedIn = Math.min(

effTotalIn,

Math.ceil((signals.cachedPrefixTokens ?? 0) \*
entry.tokenizerMultiplier)

);

const effFreshIn = effTotalIn - effCachedIn;

const effOut = Math.ceil(signals.expectedOutputTokens \*
entry.tokenizerMultiplier);

if (entry.longPromptCliffTokens && effTotalIn \>
entry.longPromptCliffTokens) {

inRate = entry.longPromptInputPer1M ?? inRate;

outRate = entry.longPromptOutputPer1M ?? outRate;

}

const batchMult = useBatchApi ? entry.batchDiscountFactor : 1.0;

const inputCost = ((effFreshIn \* inRate + effCachedIn \* cacheRate) /
1_000_000) \* batchMult;

const outputCost = ((effOut \* outRate) / 1_000_000) \* batchMult;

return Number((inputCost + outputCost).toFixed(6));

}

escalateOnNumericContradiction: true,

.filter((entry): entry is ModelCatalogEntry =\> {

effort === 'high' ? 16_000 : effort === 'medium' ? 6_000 : effort ===
'low' ? 1_500 : undefined,

useBatchApi: chosen.batchAllowed,

maxOutputTokens: Math.min(

chosen.entry.maxOutputTokens,

Math.max(512, Math.ceil(signals.expectedOutputTokens \*
chosen.entry.tokenizerMultiplier \* 1.25))

),

estimatedCostUsd: chosen.cost,

cascadePolicy,

telemetry: {

complexityScore,

entityDensityPer1kTokens,

detectedEntities,

effectiveInputTokens: Math.ceil(signals.estimatedInputTokens \*
chosen.entry.tokenizerMultiplier),

budgetCapExceeded,

};
